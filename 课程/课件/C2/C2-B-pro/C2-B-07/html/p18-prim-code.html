void Prim(ALGraph &G){
bool include[MaxVertexNum];
int dist[MaxVertexNum], parent[MaxVertexNum];
for(int i = 0; i < G.vexnum; i++)
{ include[i] = false; dist[i] = INF;
parent[i] = -1; } // 不能写 dist[] = {INF}
dist[0] = 0;
for(int k = 0; k < G.vexnum; k++){
int u = -1, minDist = INF;
for(int i = 0; i < G.vexnum; i++) // ①选树外 dist 最小的 u
if(!include[i] && dist[i] < minDist)
{ minDist = dist[i]; u = i; }
if(u == -1) break; // 图不连通，停
include[u] = true; // ②u 并入树
for(ArcNode *p = G.vertices[u].firstarc;
p != NULL; p = p->nextarc)
if(!include[p->adjvex] && p->weight < dist[p->adjvex])
{ dist[p->adjvex] = p->weight; parent[p->adjvex] = u; } // ③刷新
}
}
