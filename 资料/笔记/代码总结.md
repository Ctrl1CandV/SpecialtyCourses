# 全局约定（阅读前必看）

802考的是伪代码，重思路轻语法，但**约定必须自己前后统一**，否则改一行就全错。本文统一如下：

1. **下标基准分两套，按章节区分**：
   - 线性表/查找章节：数据从 `data[0]` 开始存，第 i 个元素是 `data[i-1]`；
   - 排序章节：`data[0]` 留作哨兵/暂存位，数据从 `data[1]` 存到 `data[length]`（这是教材惯例，堆排序要求 `2i` 是左孩子，必须从1开始）。
   每道题开头先写一句"本题下标从0/1起"，阅卷不会挑刺，但自相矛盾会扣分。
2. **链表默认带头结点**，`L` 指向头结点，`L->next` 是第一个数据结点；不带头结点的变体单独标注。
3. **顺序栈 `top` 初值为 -1**（`S.data[++S.top]` 入栈）；**循环队列牺牲一个单元**判满（`(rear+1)%MaxSize==front`）。
4. **栈/队列作为工具使用时，元素类型按需替换**：本文 `SqStack`/`SqQueue` 的 `data` 写成 `int`，但在树、图算法里存放的是 `BiTree`、`LNode *`、顶点编号等。考场上直接写 `SqStack S; Push(S, p);` 即可，无需重新定义，**但要在答题时说明一句"栈中存放结点指针"**。
5. **`int a[n] = {0}` 是伪代码简写**：C语言中变长数组不允许带初始化列表，严格写法是先声明再用循环赋初值。802允许这样简写，但**赋非0初值（如 `{INF}`）时必须写循环**——`{INF}` 只会初始化第一个元素，其余为0，这是真实的逻辑错误。
6. 参数用 C++ 引用 `&` 表示"传入传出"，等价于 C 的二级指针，伪代码中允许。

```c++
}
```

```c++
bool IsPalindrome(LinkList L, int n){
```

#### 归并两个递增表为递减单链表

```c++
bool Union(LinkList &la, LinkList lb){
    if(la == NULL || lb == NULL) return false;      // 任一为空就不能解引用，必须用||
    LNode *pa = la->next, *pb = lb->next;
    la->next = NULL;						// 以la作为新链表
    while(pa != NULL && pb != NULL){
        LNode *temp;
        if(pa->data <= pb->data){
            temp = pa;
            pa = pa->next;
        }else{
            temp = pb;
            pb = pb->next;
        }
        temp->next = la->next;
        la->next = temp;
    }
    while(pa != NULL){
        LNode *temp = pa;
        pa = pa->next;
        temp->next = la->next;
        la->next = temp;
    }
    while(pb != NULL){
        LNode *temp = pb;
        pb = pb->next;
        temp->next = la->next;
        la->next = temp;
    }
    free(lb);                               // lb的头结点已无用，顺手释放
    return true;
}
```

> 思路要点：**归并成递减表，用头插即可自然反序**，不需要先归并再逆置。这也是"递增表归并成递减表"这一类题的通用套路。
#### 最快合并两个循环单链表
```c++
/* 核心思想：将较短表的尾指向较长表的第一个元素，再将较长表的头指向较短表的第一个元素，最后释放较短表头结点 */
bool Union(LinkList L1, LinkList L2, int m, int n){
    if(m <= 0 || n <= 0) return false;
    if(m < n){
        LNode *p = L1->next;
        while(p->next != L1) p = p->next;
        p->next = L2->next;
        L2->next = L1->next;
        free(L1);
    }else{
        LNode *p = L2->next;
        while(p->next != L2) p = p->next;
        p->next = L1->next;
        L1->next = L2->next;
        free(L2);
    }
    return true;
}
```

#### 408真题19年41题

设线性表L=(a1,a2,a3,...,an)采用带头结点的单链表存储，现在需要设计一个空间复杂度为O(1)且时间上尽可能高效的算法，重新排列L中的各结点，得到线性表L'=(a1,an,a2,an-1,...)

```c++
typedef struct node{
    int data;
    struct node *next;
}NODE;
void Reorder(NODE *head){
    if(head == NULL || head->next == NULL || head->next->next == NULL) return;
    NODE *slow = head->next, *fast = head->next;
    while(fast != NULL && fast->next != NULL){
        slow = slow->next;
        fast = fast->next->next;
    }

    NODE *second = slow->next, *p = second->next;
    slow->next = NULL;
    while(p != NULL){
        NODE *temp = p->next;
        p->next = second->next;
        second->next = p;
        p = temp;
    }

    NODE *first = head->next;
    while(second != NULL){
        NODE *temp1 = first->next, *temp2 = second->next;
        second->next = temp1;first->next = second;
        first = temp1;second = temp2;
    }
}
```

# 栈和队列

栈和队列是较为经典的算法工具，在很多场景下都需要借助这二者。作为工具使用时，一般只需进行简单的初始化和基础操作，整体考察难度不大，但要注意一些定义细节，例如栈顶指针的初值是 top=0 还是 top=-1。

```c++
int strassign(String &str, char *ch) {

    char *c = ch;

    if(len == 0){                                   // 空串则赋值NULL和0

        str.length = 0;
```

- 非递归后序遍历
```c++
void PostOrderTraverse(BiTree T){
    Stack s1, s2;InitStack(s1);InitStack(s2);
    BiTNode *p;
    Push(s1, T);
    // 用前序的方式遍历，将遍历序列按序存入栈中逆置读取，得到后序遍历序列
    while(!IsEmpty(s1)){
        Pop(s1, p);Push(s2, p);
        if(p->lchild) Push(s1, p->lchild);
        if(p->rchild) Push(s1, p->rchild);
    }
    while(!IsEmpty(s2)){
        Pop(s2, p);
        visit(p);
    }
}
```

```c++
        p = Q[front++];temp++;

        if(p->rchild != NULL) Q[++rear] = p->rchild;

            if(temp > maxwidth) maxwidth = temp;
            temp = 0;
        }
    }
    return maxwidth;
}
```

```c++
void Prim(ALGraph &G) {
    bool include[G.vexnum];
    int dist[G.vexnum], parent[G.vexnum];                       // parent记录生成树中该点的父结点
    // 注意：不能写 int dist[] = {INF}，那样只有第一个元素是INF，其余为0

        dist[i] = INF;
        parent[i] = -1;

    for (int k = 0; k < G.vexnum; k++){
        int u = -1, minDist = INF;
        for (int i = 0; i < G.vexnum; i++){
            if (!include[i] && dist[i] < minDist){              // 寻找当前不在生成树内且距离最小的邻接点
                minDist = dist[i];
                u = i;
            }
        }
        if (u == -1) break;

        include[u] = true;                                      // 将寻找到的点入队并更新dist
        for (ArcNode *p = G.vertices[u].firstarc; p != NULL; p = p->nextarc) {
            int v = p->adjvex, w = p->weight;
            if (!include[v] && w < dist[v]) {
                dist[v] = w;
                parent[v] = u;
            }
        }
    }
```

```c++
void TwoRoadSort(SqList &L){
    int n = L.length;
    int temp[MaxSize];                          // 当作容量为n的"循环数组"使用，下标0 ~ n-1
    int first = 0, final = 0;                   // first指向最小值位置，final指向最大值位置
    temp[0] = L.data[1];                        // 原表数据存于L.data[1..n]，先放入第一个元素
    for (int i = 2; i <= n;i++){
        // 待插入元素比最小元素还小：向first前方（逆时针）扩展
        if (L.data[i] < temp[first]){
            first = (first - 1 + n) % n;
            temp[first] = L.data[i];
        }
        // 待插入元素比最大元素还大：向final后方（顺时针）扩展
        else if (L.data[i] > temp[final]){
            final = (final + 1) % n;
            temp[final] = L.data[i];
        }
        // 待插入元素落在中间：从final一侧向后腾位置，再插入
        else {
            int k = (final + 1) % n;
            while (temp[(k - 1 + n) % n] > L.data[i]) {
                temp[k] = temp[(k - 1 + n) % n];
                k = (k - 1 + n) % n;
            }
            temp[k] = L.data[i];
            final = (final + 1) % n;
        }
    }
    // 从first开始按环序回填到L.data[1..n]，与本章"1起存"约定一致
    for (int k = 0; k < n;k++) {
        L.data[k + 1] = temp[(first + k) % n];
    }
}
```

```c++
void QuickSort(int R[], int low, int high){
    if(low >= high) return;
    int sum = 0;
    for(int i = low;i <= high;i++) sum += R[i];      // 只对当前待划分区间求和
    int avg = sum / (high - low + 1);
    int i = low, j = high;
    while(i <= j){
        // 以阈值做双指针划分，交换不满足阈值的一对元素

            R[i] = R[j];
            R[j] = temp;
            i++; j--;
        }
    }

}
```

```c++
int Josephus(int n, int s, int m){

    head->data = 1;
    LNode *r = head;
    for(int i = 2;i <= n;i++){
        LNode *p = (LNode*)malloc(sizeof(LNode));
        p->data = i;
        r->next = p;
        r = p;
```

```c++

int MergeSort(int A[], int temp[], int left, int right){
    if(left >= right) return 0;
    int mid = (left + right) / 2, count = 0;
    count += MergeSort(A, temp, left, mid);
    count += MergeSort(A, temp, mid + 1, right);

}
int merge(int A[], int temp[], int left, int mid, int right){
    for(int k = left;k <= right;k++) temp[k] = A[k];        // 先复制到辅助表
    int i = left, j = mid + 1, k = left, count = 0;
    while(i <= mid && j <= right){
        if(temp[i] <= temp[j]) A[k++] = temp[i++];
        else{
            count += mid - i + 1;                           // 左半剩余元素均大于temp[j]，构成逆序对
            A[k++] = temp[j++];
        }
    }
    while(i <= mid) A[k++] = temp[i++];
    while(j <= right) A[k++] = temp[j++];
    return count;
```
