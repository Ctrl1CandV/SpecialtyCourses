### 缺口 1：头插建表、尾插建表、求表长、销毁链表（C2-A-03）

材料现状：`代码总结.md` 有单链表的初始化、GetElem、LocateElem、ListInsert、ListDelete，但没有建表、求表长与销毁；`重邮数据结构.md` 有头插尾插的文字描述但无完整代码。

```c
// 头插法建表：读入 n 个数，结果是输入次序的逆序
void CreateByHead(LinkList &L, int n){
    L = (LNode *)malloc(sizeof(LNode));
    if(L == NULL) return;
    L->next = NULL;                       // 先成空表
    for(int i = 0; i < n; i++){
        LNode *s = (LNode *)malloc(sizeof(LNode));
        if(s == NULL) return;
        scanf("%d", &s->data);
        s->next = L->next;                // 新结点接在原第一个之前
        L->next = s;
    }
}

// 尾插法建表：结果与输入次序一致，rear 让每次插入 O(1)
void CreateByTail(LinkList &L, int n){
    L = (LNode *)malloc(sizeof(LNode));
    if(L == NULL) return;
    LNode *r = L;                         // r 始终指向当前尾结点
    for(int i = 0; i < n; i++){
        LNode *s = (LNode *)malloc(sizeof(LNode));
        if(s == NULL) return;
        scanf("%d", &s->data);
        r->next = s;
        r = s;
    }
    r->next = NULL;                       // 尾结点的 next 必须置空，否则是垃圾值
}

// 求表长：不含头结点
int ListLength(LinkList L){
    int n = 0;
    LNode *p = L->next;
    while(p != NULL){ n++; p = p->next; }
    return n;
}

// 销毁链表：先存后继再 free，否则读的是已回收内存
void DestroyList(LinkList &L){
    LNode *p = L, *q;
    while(p != NULL){
        q = p->next;
        free(p);
        p = q;
    }
    L = NULL;                             // 表已不存在，头指针置空
}
```

教学要点：头插的两行次序不能换；尾插不留 rear 会从 O(n) 退化成 O(n²)；销毁必须先存后继。

### 缺口 2：静态链表的定义与插入删除（C2-A-04）

材料现状：`重邮数据结构.md` 有静态链表的知识描述，两份材料都无代码。

```c
#define MaxSize 100
typedef struct{
    int data;
    int next;                    // 游标：下一个元素的下标，-1 表示结束
}Component;

Component L[MaxSize];
int freelist;                    // 备用链的头下标

// 初始化：下标 0 作数据链的头结点，1..MaxSize-1 串成备用链
void InitSLink(){
    L[0].next = -1;
    for(int i = 1; i < MaxSize - 1; i++) L[i].next = i + 1;
    L[MaxSize - 1].next = -1;
    freelist = 1;
}

// 从备用链取一个下标（相当于 malloc）
int SLinkMalloc(){
    int s = freelist;
    if(s != -1) freelist = L[s].next;
    return s;                    // 返回 -1 表示没有空闲位置
}

// 归还一个下标（相当于 free）
void SLinkFree(int s){
    L[s].next = freelist;
    freelist = s;
}

// 在第 i 个元素之前插入
bool InsertSLink(int i, int elem){
    int p = 0, j = 0;
    while(p != -1 && j < i - 1){ p = L[p].next; j++; }   // 找第 i-1 个
    if(p == -1) return false;
    int s = SLinkMalloc();
    if(s == -1) return false;
    L[s].data = elem;
    L[s].next = L[p].next;
    L[p].next = s;
    return true;
}

// 删除第 i 个元素，值用 elem 带出
bool DeleteSLink(int i, int &elem){
    int p = 0, j = 0;
    while(p != -1 && j < i - 1){ p = L[p].next; j++; }
    if(p == -1 || L[p].next == -1) return false;
    int s = L[p].next;
    elem = L[s].data;
    L[p].next = L[s].next;
    SLinkFree(s);
    return true;
}
```

教学要点：next 里存的是下标不是地址，所以叫游标；插入删除只改游标不搬数据；备用链就是"空闲位置的链表"，SLinkMalloc 与 SLinkFree 对应 malloc 与 free。

### 缺口 3：链栈与链队（C2-A-05）

材料现状：`重邮数据结构.md` 有链栈的知识（并说明链队一般不带头结点），两份材料都无代码。

**口径偏离声明（必须写在课件页面上）**：`重邮数据结构.md` 规定链栈**没有头结点**（Lhead 直接指栈顶元素），而下面的补写版用了**带头结点**的单链表。这是有意为之：遵循课程全局约定 2（链表默认带头结点），与 C2-A-03 的单链表、C2-A-05 的链队判空口径保持一致，学生不用切换体系。但学生手里的笔记与真题可能用无头结点版，所以课件要明说两种写法的差别：带头结点时栈空是 `S->next == NULL`，无头结点时栈空是 `S == NULL`；程序阅读题遇到无头结点版要能认出来。

```c
// 链栈：带头结点的单链表，栈顶就是表头，入栈等于头插
typedef struct LNode{
    int data;
    struct LNode *next;
}LNode, *LinkStack;

bool InitLinkStack(LinkStack &S){
    S = (LNode *)malloc(sizeof(LNode));
    if(S == NULL) return false;
    S->next = NULL;
    return true;
}
bool PushL(LinkStack &S, int x){
    LNode *s = (LNode *)malloc(sizeof(LNode));
    if(s == NULL) return false;               // 判空不能省
    s->data = x;
    s->next = S->next;
    S->next = s;
    return true;
}
bool PopL(LinkStack &S, int &x){
    if(S->next == NULL) return false;         // 栈空
    LNode *s = S->next;
    x = s->data;
    S->next = s->next;
    free(s);
    return true;
}
bool StackEmptyL(LinkStack S){ return S->next == NULL; }

// 链队：不带头结点，front 与 rear 两个指针
typedef struct{
    LNode *front, *rear;
}LinkQueue;

bool InitLinkQueue(LinkQueue &Q){ Q.front = Q.rear = NULL; return true; }

bool EnQueueL(LinkQueue &Q, int x){
    LNode *s = (LNode *)malloc(sizeof(LNode));
    if(s == NULL) return false;
    s->data = x; s->next = NULL;
    if(Q.rear == NULL){ Q.front = Q.rear = s; }   // 原队空，两个指针都指新结点
    else{ Q.rear->next = s; Q.rear = s; }
    return true;
}

bool DeQueueL(LinkQueue &Q, int &x){
    if(Q.front == NULL) return false;             // 队空
    LNode *s = Q.front;
    x = s->data;
    Q.front = s->next;
    if(Q.front == NULL) Q.rear = NULL;            // 删到只剩一个时必须同时更新 rear
    free(s);
    return true;
}
bool QueueEmptyL(LinkQueue Q){ return Q.front == NULL; }
```

教学要点：链栈不需要判满（内存够就能压）；链队不带头结点时队空是 `front == NULL && rear == NULL`，出队到只剩一个时要同时把 rear 置空，否则 rear 变成悬空指针。

### 缺口 4：括号匹配、中缀转后缀、后缀求值、双栈模拟队列（C2-B-03）

材料现状：`重邮数据结构.md` 有括号匹配与表达式求值的知识描述，两份材料都无完整代码。

```c
// 括号匹配：三种失配情况
bool BracketMatch(char str[], int n){
    SqStack S; InitStack(S);
    for(int i = 0; i < n; i++){
        if(str[i] == '(' || str[i] == '[' || str[i] == '{') Push(S, str[i]);
        else if(str[i] == ')' || str[i] == ']' || str[i] == '}'){   // 只处理右括号，其他字符跳过
            if(IsEmpty(S)) return false;                  // ① 右括号多了
            char top; Pop(S, top);
            if((top == '(' && str[i] != ')') ||
               (top == '[' && str[i] != ']') ||
               (top == '{' && str[i] != '}')) return false;  // ② 类型不匹配（交叉）
        }
    }
    return IsEmpty(S);                                    // ③ 栈非空说明左括号多了
}
```

**为什么必须写成 `else if` 而不是 `else`**：用 `else` 会把任何非左括号的字符（字母、数字、运算符）都当成右括号处理，输入一个带操作数的表达式如 `a+(b*c)` 时，扫到 'a' 就会弹栈并判失配，结果全错。这一处要在课堂上当反例讲：**处理字符序列时，“不是我要的左括号”与“是我要的右括号”是两件事，不能用 else 一笔带过**。

```c
// 中缀转后缀：Prior(c) 返回运算符优先级，( 在栈内优先级最高、在栈外最低
void ToPostfix(char infix[], char postfix[]){
    SqStack S; InitStack(S);
    int k = 0;
    for(int i = 0; infix[i] != '\0'; i++){
        char c = infix[i];
        if(c >= '0' && c <= '9') postfix[k++] = c;                 // 操作数直接输出
        else if(c == '(') Push(S, c);
        else if(c == ')'){
            char top;
            while(GetTop(S, top) && top != '('){ Pop(S, top); postfix[k++] = top; }
            Pop(S, top);                                            // 弹出左括号，不输出
        }else{
            char top;
            while(GetTop(S, top) && top != '(' && Prior(top) >= Prior(c)){
                Pop(S, top); postfix[k++] = top;
            }
            Push(S, c);
        }
    }
    char top;
    while(!IsEmpty(S)){ Pop(S, top); postfix[k++] = top; }
    postfix[k] = '\0';
}

// 后缀求值：先弹出的是右操作数
int EvalPostfix(char postfix[]){
    SqStack S; InitStack(S);
    for(int i = 0; postfix[i] != '\0'; i++){
        char c = postfix[i];
        if(c >= '0' && c <= '9'){ int x = c - '0'; Push(S, x); }
        else{
            int b, a; Pop(S, b); Pop(S, a);
            switch(c){
                case '+': Push(S, a + b); break;
                case '-': Push(S, a - b); break;
                case '*': Push(S, a * b); break;
                case '/': Push(S, a / b); break;
            }
        }
    }
    int result; Pop(S, result);
    return result;
}

// 双栈模拟队列：s1 负责入，s2 负责出，s2 空了才把 s1 全倒过去
SqStack s1, s2;
bool EnQueue2(int x){ return Push(s1, x); }
bool DeQueue2(int &x){
    if(!IsEmpty(s2)) return Pop(s2, x);
    while(!IsEmpty(s1)){ int t; Pop(s1, t); Push(s2, t); }   // 倒一次，顺序反一次
    if(IsEmpty(s2)) return false;
    return Pop(s2, x);
}
```

教学要点：中缀转后缀的规则 ④ 是"大于等于"（同级先弹，这就是左结合）；后缀求值先弹出的是右操作数（栈后进先出），减法除法搞反就全错；双栈模拟队列只有 s2 空了才倒，否则顺序会乱；**括号匹配必须用 `else if` 只拦右括号**（理由见上方代码注）。

**待核对**：Prior 函数的具体实现（左括号在栈内与栈外的优先级不同）制作课件时按教材口径写全，本文件只给调用处。

### 缺口 5：BST 的删除（C2-B-08）

材料现状：`重邮数据结构.md` 有 BST 的查找与插入代码，两份材料都没有删除。删除是 BST 三个操作里唯一的难点，802 的程序阅读题与综合应用题都可能考。

```c
typedef struct BSTNode{
    int key;
    struct BSTNode *lchild, *rchild;
}BSTNode, *BSTree;

// 参数是指针的引用：要改的是父结点的 lchild 或 rchild 字段
int BSTDelete(BSTree &T, int key){
    if(T == NULL) return 0;                                  // 没找到
    if(key < T->key) return BSTDelete(T->lchild, key);
    if(key > T->key) return BSTDelete(T->rchild, key);
    // 找到待删结点，分三种情况
    if(T->lchild == NULL && T->rchild == NULL){              // ① 叶子
        free(T); T = NULL;
        return 1;
    }
    if(T->lchild == NULL){                                   // ② 只有右孩子
        BSTNode *s = T; T = T->rchild; free(s);
        return 1;
    }
    if(T->rchild == NULL){                                   // ② 只有左孩子
        BSTNode *s = T; T = T->lchild; free(s);
        return 1;
    }
    // ③ 两个孩子：用中序前驱（左子树里最大的）替换，再递归删那个前驱
    BSTNode *pre = T->lchild;
    while(pre->rchild != NULL) pre = pre->rchild;
    T->key = pre->key;
    return BSTDelete(T->lchild, pre->key);
}
```

教学要点：情况 ③ 不能直接删，因为删完两棵子树就断了，所以先用一个值替换再删那个值的结点；替换值取中序前驱或中序后继都行，前驱一定没有右孩子、后继一定没有左孩子，于是回到情况 ① 或 ②。为什么用中序前驱能保持 BST 性质：前驱是中序序列里刚好小于它的值，替换后左子树全部小于它、右子树全部大于它。

### 缺口 6：AVL 的插入与四种旋转（C2-B-08）

材料现状：`重邮数据结构.md` 有四种旋转的文字描述与图（"让 B 上位当根""让 C 上位当根"），无代码。802 对 AVL 考的是手算（插入后画树、指出旋转类型），代码不是考点，所以课件只给指针操作的代码，bf 的维护标注为待核对。

```c
typedef struct AVLNode{
    int data;
    int bf;                          // 平衡因子：左子树高度减右子树高度
    struct AVLNode *lchild, *rchild;
}AVLNode, *AVLTree;

// LL 型：右单旋
void RotateR(AVLTree &T){
    AVLTree B = T->lchild;
    T->lchild = B->rchild;           // B 的右子树回挂给 T 当左子树
    B->rchild = T;
    T->bf = 0; B->bf = 0;            // 单旋后两个结点的 bf 都归 0
    T = B;                           // B 上位当根
}

// RR 型：左单旋（与右单旋对称）
void RotateL(AVLTree &T){
    AVLTree B = T->rchild;
    T->rchild = B->lchild;
    B->lchild = T;
    T->bf = 0; B->bf = 0;
    T = B;
}

// LR 型：先对 B 左旋，再对 T 右旋
void RotateLR(AVLTree &T){
    AVLTree B = T->lchild, C = B->rchild;
    B->rchild = C->lchild;           // C 的左子树回挂给 B 当右子树
    C->lchild = B;
    T->lchild = C->rchild;           // C 的右子树回挂给 T 当左子树
    C->rchild = T;
    // bf 的三种情况按 C 旋转前的 bf 定，见下方"待核对"
    T = C;                           // C 上位当根
}

// RL 型：先对 B 右旋，再对 T 左旋（与 LR 对称）
void RotateRL(AVLTree &T){
    AVLTree B = T->rchild, C = B->lchild;
    B->lchild = C->rchild;
    C->rchild = B;
    T->rchild = C->lchild;
    C->lchild = T;
    T = C;
}
```

**待核对（制作课件前必须做）**：双旋后 A（即 T）与 B 的 bf 取决于 C 旋转前的 bf，共三种情况。bf 的符号约定（左减右还是右减左）不同，三种情况的取值也不同。制作课件前用教材（严蔚敏或王道）核对一遍，把三种情况的 bf 赋值补进 RotateLR 与 RotateRL，并在页面上写清本课用的符号约定。**不要在没核对的情况下把 bf 表放上页面。**

### 缺口 7：散列表的构造与两种冲突处理（C2-B-08）

材料现状：`重邮数据结构.md` 有 22 处散列相关知识（构造方法、处理冲突的方法、查找与性能分析），无代码。

```c
#define TableLen 13                    // 表长
#define EMPTY -1                       // 空位标记（要求关键字都不等于 -1）
#define DELETED -2                     // 删除标记：开放定址不能直接置 EMPTY

int Hash(int key){ return key % 11; }  // 除留余数法，p 取不大于表长的最大质数

// 开放定址：线性探测
void InsertOpen(int H[], int key){
    int addr = Hash(key);
    while(H[addr] != EMPTY && H[addr] != DELETED){
        if(H[addr] == key) return;                     // 已存在，不重复插
        addr = (addr + 1) % TableLen;                  // 线性探测下一位
    }
    H[addr] = key;
}
int SearchOpen(int H[], int key){
    int addr = Hash(key);
    while(H[addr] != EMPTY){                           // 遇到 EMPTY 才停，DELETED 要继续探
        if(H[addr] == key) return addr;
        addr = (addr + 1) % TableLen;
    }
    return -1;
}
void DeleteOpen(int H[], int key){
    int addr = SearchOpen(H, key);
    if(addr != -1) H[addr] = DELETED;                  // 只能打删除标记
}

// 链地址：同义词拉成一条单链表，H 是 TableLen 个链头指针的数组
void InsertLink(LinkList H[], int key){
    int addr = Hash(key);
    LNode *s = (LNode *)malloc(sizeof(LNode));
    if(s == NULL) return;
    s->data = key;
    s->next = H[addr];                                 // 头插到同义词链
    H[addr] = s;
}
int SearchLink(LinkList H[], int key){
    int addr = Hash(key);
    for(LNode *p = H[addr]; p != NULL; p = p->next)
        if(p->data == key) return addr;
    return -1;
}
```

教学要点：**开放定址的删除不能直接置 EMPTY**，因为会断开探测链（后面靠探测找到的元素就再也找不到了），必须用专门的删除标记，且查找时遇到删除标记要继续探测。这一点是学生最容易错的，也是选择题的考点。链地址不需要删除标记，直接摘结点。

**SearchOpen 的一个边界要注明**：`while(H[addr] != EMPTY)` 的终止依赖表里至少有一个 EMPTY 槽。如果表满（装载因子 = 1）且查一个不存在的关键字，这个循环会无限探测。教材通用前提是装载因子小于 1，课件在页面上注一句“本函数在表不满（至少有一个空位）时成立”，或加一个探测次数上限 `for(int cnt = 0; cnt < TableLen; cnt++)` 兜底。

ASL 的计算训练留给专业课（题集 9答26、9答27、9答28），本课只要求能写构造与查找代码、能说出两种方法的代价差别。

### 缺口 8：哈夫曼树的构造（C2-B-05，只给思想与骨架）

材料现状：`重邮数据结构.md` 有哈夫曼树（高频）、构造、哈夫曼编码三节知识，无代码。802 以手算为主（题集 6选33、6选34、6选36、6选40、6答26），课件只给思想与骨架，课后不留代码题。

```c
// 用 2n-1 个结点的数组存哈夫曼树，三个字段记录父子关系（严蔚敏教材的经典写法）
typedef struct{
    int weight;
    int parent, lchild, rchild;      // 存下标，0 表示没有
}HTNode;

void CreateHuffman(HTNode HT[], int w[], int n){
    int m = 2 * n - 1;                            // n 个叶子共需 2n-1 个结点
    for(int i = 1; i <= n; i++){                  // 前 n 个是叶子
        HT[i].weight = w[i];
        HT[i].parent = HT[i].lchild = HT[i].rchild = 0;
    }
    for(int i = n + 1; i <= m; i++)               // 后 n-1 个待生成
        HT[i].weight = HT[i].parent = HT[i].lchild = HT[i].rchild = 0;

    for(int i = n + 1; i <= m; i++){              // 建 n-1 个内部结点
        int s1, s2;
        SelectTwoMin(HT, i - 1, s1, s2);          // 在前 i-1 个里选 parent==0 的两个最小
        HT[s1].parent = i;
        HT[s2].parent = i;
        HT[i].lchild = s1;
        HT[i].rchild = s2;
        HT[i].weight = HT[s1].weight + HT[s2].weight;
    }
}
```

教学要点：为什么是 2n-1 个结点（n 个叶子，每次合并减少一棵树，合并 n-1 次）；为什么用 parent 字段为 0 表示"还没被合并"；SelectTwoMin 是每轮扫一遍找两个最小，所以构造是 O(n²)，用优先队列可以降到 O(nlogn)（提一句，不展开）。**课件只给到这一层，编码部分（左右分支标 0 与 1）用手算讲**。

### 缺口 9：Kruskal（含并查集）、Dijkstra、Floyd（C2-B-07）

材料现状：`代码总结.md` 有 Prim；`重邮数据结构.md` 有 Kruskal（中频）、Dijkstra（中频）、Floyd 的知识与手算表，三者都无代码。

```c
// ---------- 并查集（A9 埋的钩子在此兑现）----------
int Find(int parent[], int x){
    while(parent[x] != x) x = parent[x];        // 沿父结点往上找到根
    return x;
}
void Union(int parent[], int a, int b){
    int ra = Find(parent, a), rb = Find(parent, b);
    if(ra != rb) parent[ra] = rb;               // 一棵树的根接到另一棵的根下
}

// ---------- Kruskal ----------
typedef struct{ int u, v, w; }Edge;             // 一条边：两端点与权值

void Kruskal(ALGraph G, Edge E[], int m, int vexnum){
    // 前提：E 里已按 w 从小到大排好序（伪代码里一句"按权值排序"带过）
    int parent[MaxVertexNum];
    for(int i = 0; i < vexnum; i++) parent[i] = i;   // 每个顶点自成一棵树
    int cnt = 0;
    for(int i = 0; i < m && cnt < vexnum - 1; i++){
        int a = Find(parent, E[i].u), b = Find(parent, E[i].v);
        if(a != b){                                  // 不在同一棵树，加入不成环
            parent[a] = b;
            cnt++;
        }
    }
}

// ---------- Dijkstra（单源最短路径，邻接矩阵）----------
void Dijkstra(MGraph G, int v0, int dist[], int path[], bool S[]){
    const int INF = 0x3f3f3f3f;
    for(int i = 0; i < G.vexnum; i++){
        dist[i] = G.edge[v0][i];                     // 初始：v0 直接可达的距离
        S[i] = false;
        path[i] = (G.edge[v0][i] < INF) ? v0 : -1;
    }
    dist[v0] = 0; S[v0] = true;
    for(int k = 1; k < G.vexnum; k++){
        int u = -1, minDist = INF;
        for(int i = 0; i < G.vexnum; i++)
            if(!S[i] && dist[i] < minDist){ minDist = dist[i]; u = i; }
        if(u == -1) break;                           // 剩下的都不可达
        S[u] = true;                                 // u 的最短距离已确定
        for(int i = 0; i < G.vexnum; i++)            // 用 u 松弛其余顶点
            if(!S[i] && G.edge[u][i] < INF && dist[u] + G.edge[u][i] < dist[i]){
                dist[i] = dist[u] + G.edge[u][i];
                path[i] = u;
            }
    }
}

// ---------- Floyd（各点对最短路径，邻接矩阵）----------
void Floyd(MGraph G, int A[][MaxVertexNum]){
    for(int i = 0; i < G.vexnum; i++)
        for(int j = 0; j < G.vexnum; j++) A[i][j] = G.edge[i][j];   // 初始化

    for(int k = 0; k < G.vexnum; k++)              // k 必须在最外层
        for(int i = 0; i < G.vexnum; i++)
            for(int j = 0; j < G.vexnum; j++)
                if(A[i][k] + A[k][j] < A[i][j]) A[i][j] = A[i][k] + A[k][j];
}
```

教学要点：

- Kruskal 靠并查集判环：一条边的两端已在同一棵树里，加入就成环。时间 O(e log e)（排序主导），所以稀疏图用 Kruskal、稠密图用 Prim。
- **Dijkstra 与 Prim 的骨架几乎一样**，差别只在更新式：Prim 用 `w < dist[v]`（只看边权，要的是到生成树的最短边），Dijkstra 用 `dist[u] + w < dist[i]`（要累加路径长度）。把这两行并排放，学生就能记住两个算法的区别。
- **Floyd 的 k 必须在最外层**：A[i][k] 与 A[k][j] 必须是"只允许用前 k-1 个顶点做中转"的结果，k 放内层会用到本轮还没算完的值。这是学生最容易写错的一处。
- Floyd 用 INF 相加可能溢出，实际实现要判 `A[i][k] < INF && A[k][j] < INF`；伪代码里可以省略，但课件要提一句（0x3f3f3f3f 的设计就是为了让两个 INF 相加不溢出成负数）。

### 缺口 13：2023 像素距离图（多源 BFS，C3-B-04）

材料现状：`代码总结.md` 真题汇总里 2023 第 2 题只有题面，无解答。

```c
// 题目：n×m 的 01 位图，1 表示白像素，求每个像素到最近白像素的曼哈顿距离
// 关键思路：从所有白像素同时往外扩散（多源 BFS），而不是对每个黑像素各做一次 BFS
#define MaxN 1000
#define MaxM 1000
int dir[4][2] = {{-1,0},{1,0},{0,-1},{0,1}};      // 上下左右

void PixelDist(int g[][MaxM], int dist[][MaxM], int n, int m){
    SqQueue Q; InitQueue(Q);
    // 第一步：所有白像素先全部入队、dist 置 0；其余置 -1 表示未访问
    for(int i = 0; i < n; i++)
        for(int j = 0; j < m; j++){
            if(g[i][j] == 1){ dist[i][j] = 0; EnQueue(Q, i * m + j); }
            else dist[i][j] = -1;
        }
    // 第二步：普通 BFS 逐层扩散，每扩一层距离加 1
    while(!IsEmpty(Q)){
        int cur; DeQueue(Q, cur);
        int x = cur / m, y = cur % m;              // 一维编号还原为二维坐标
        for(int d = 0; d < 4; d++){
            int nx = x + dir[d][0], ny = y + dir[d][1];
            if(nx < 0 || nx >= n || ny < 0 || ny >= m) continue;    // 越界
            if(dist[nx][ny] == -1){                                  // 未访问，dist 兼作 visited
                dist[nx][ny] = dist[x][y] + 1;
                EnQueue(Q, nx * m + ny);
            }
        }
    }
}
// 时间复杂度 O(nm)：每个像素入队出队各一次；空间 O(nm)：队列最坏存一层
```

教学要点：与普通 BFS **只差第一步**（所有源点先全部入队且 dist 置 0），其余完全一样。因为 BFS 逐层扩散的性质，每个黑像素第一次被访问时的 dist 就是它到最近白像素的距离。三个细节：队列里存一维编号 `i*m+j`（材料的 SqQueue 只存一个 int，取出时用除与模还原）；`dist == -1` 同时充当 visited，省一个数组；四个方向用数组避免写四段重复代码。

答题时的思想描述可以这样写：直接对每个黑像素做一次 BFS 是 O(n²m²)，会超时；反过来从所有白像素同时开始 BFS 逐层扩散，每个像素第一次被访问到的层号就是它到最近白像素的距离，时间 O(nm)。

### 缺口 14：2022 稀疏多项式奇偶分解（C2-B-02、C3-B-02）

```c
// 题目：循环链表表示的稀疏多项式，分解成两个链，各自只含奇次项或偶次项，
//       要求利用原链表中的结点空间（数据结点不许新 malloc）
typedef struct PolyNode{
    float coef;                       // 系数
    int exp;                          // 指数
    struct PolyNode *next;
}PolyNode;

void SplitPoly(PolyNode *L, PolyNode *&odd, PolyNode *&even){
    odd  = (PolyNode*)malloc(sizeof(PolyNode));  odd->next  = odd;   // 两个新头结点，各自成环
    even = (PolyNode*)malloc(sizeof(PolyNode));  even->next = even;

    PolyNode *p = L->next, *temp;
    while(p != L){                               // 循环链表终止条件：回到头结点
        temp = p->next;                          // 先存后继，否则改指针后找不到下一个
        if(p->exp % 2 == 0){                     // 偶次项接到 even 链
            p->next = even->next;
            even->next = p;
        }else{                                   // 奇次项接到 odd 链
            p->next = odd->next;
            odd->next  = p;
        }
        p = temp;
    }
    free(L);                                     // 原头结点已无用
}
// 时间复杂度 O(n)：每个结点只处理一次；空间 O(1)：只新建两个头结点
```

教学要点：① "利用原结点空间"意味着只新建两个头结点，数据结点全部从原链上摘；② 摘结点前必须先存后继；③ 用头插接到两个新链上，顺序会反，如果题目要求保持原顺序就改用尾插并维护两个 rear（题目没明确要求时两种都给分，答题时说明一句）；④ 原头结点最后 free。

**待核对**：exp 可能是负数或浮点数吗？题目说"稀疏多项式"，指数一般是非负整数，所以 `p->exp % 2` 可用；如果指数可能为负，C 语言的 % 对负数返回负值，判断偶数要写 `p->exp % 2 == 0`（负偶数也满足），这一点仍然成立，不用改。制作课件时用真题原文核一遍 exp 的类型。

### 缺口 15：2022 最小半径生成树（C3-B-04 选讲，只给思想）

材料现状：只有题面，无解答。难度高，按设计只给思想与解法方向，不写完整代码。

题目：从根到叶子的最大距离称为树的半径，给定无向连通图，找出半径最小的生成树。

两个解法方向：

1. **枚举根 + BFS**：以每个顶点为根做一次 BFS 生成树，算出该树的半径（BFS 的最大层号就是根到叶的最大距离），取半径最小的那棵。时间 O(n(n+e))。这个解法正确的前提是"以 v 为根的最短路径树就是以 v 为根半径最小的生成树"，因为 BFS 树里每个顶点到根的距离就是最短距离，最大层号即该根下的最小可能半径。
2. **图的中心**：先用 Floyd 求全源最短路径，然后对每个顶点 i 算 `max_j A[i][j]`（即 i 的离心率），离心率最小的顶点就是图的中心，以它为根做 BFS 生成树。时间 O(n³)。

课件处理：给这两个方向各一句话，说明考场上如果遇到这道题，先用方向 1（更好写），时间不够可以先跳过。不写完整代码，理由是它的核心动作（BFS 生成树、Floyd）都已在别处讲过，剩下的组合不是 802 的高频形态。

// 修正版：与全章 1 起口径统一
void BubbleSort(SqList &L){                     // 数据存 L.data[1..length]
    for(int i = 1; i < L.length; i++){          // 修正：从 1 开始
        bool swapped = false;
        for(int j = 1; j <= L.length - i; j++)  // 修正：从 1 到 length-i

                L.data[j + 1] = tmp;
                swapped = true;
            }
        if(!swapped) break;                     // 本趟没交换说明已有序，提前退出
    }
}

### A2：以平均值为界值的快排的死递归（C2-B-09）

问题：区间内元素全部 ≤ avg 时（例如全相等），第一个内层 while 把 i 推到 high+1，第二个内层 while 因 `i <= j` 已不成立而不执行、j 停在 high；退出外层循环后 `if(low < j)` 仍成立，递归区间 [low, high] 与原来完全一样，永远停不下来。

```c
// 修正版：加进度保护，保证递归区间真的变小
void QuickSortAvg(int R[], int low, int high){
    if(low >= high) return;
    int sum = 0;
    for(int i = low; i <= high; i++) sum += R[i];      // 只对当前待划分区间求和
    int avg = sum / (high - low + 1);
    int i = low, j = high;
    while(i <= j){
        while(i <= j && R[i] <= avg) i++;
        while(i <= j && R[j] >  avg) j--;
        if(i < j){
            int t = R[i]; R[i] = R[j]; R[j] = t;
            i++; j--;
        }
    }
    // 修正：两个条件都要求区间真的变小（j < high 且 i > low）
    if(low <= j && j < high) QuickSortAvg(R, low, j);
    if(low < i && i <= high) QuickSortAvg(R, i, high);
}
void SortAvg(int R[], int n){ QuickSortAvg(R, 0, n - 1); }
```

课件处置：用修正版，并且**把 R = {5, 5, 5} 这个反例在课堂上走一遍**，让学生自己看到不加保护时递归区间怎么原地打转。这一段是本节最有价值的教学，因为它同时教了三件事：平均值划分的写法、递归必须有进度保证这个普适道理、以及怎么用一个小反例验证自己的代码。

// 修正：邻接表的边表结点必须带 weight 字段（课件统一用这个定义，见 C2-A-09）

### A4：非递归后序遍历缺判空（C2-A-07）

问题：材料的 PostOrderTraverse 没有 `if(T == NULL) return;`，空树时 Push(s1, NULL) 之后 Pop 出 NULL，再取 `p->lchild` 会崩。同文件的非递归先序版本有这一句，后序版本漏了。

```c
void PostOrder2(BiTree T){
    if(T == NULL) return;                    // 修正：补上判空
    SqStack s1, s2; InitStack(s1); InitStack(s2);
    BiTNode *p;
    Push(s1, T);
    while(!IsEmpty(s1)){                     // 先序变体"根右左"，结果压入 s2
        Pop(s1, p); Push(s2, p);
        if(p->lchild) Push(s1, p->lchild);
        if(p->rchild) Push(s1, p->rchild);
    }
    while(!IsEmpty(s2)){ Pop(s2, p); visit(p); }   // 逆置读出即"左右根"
}
```

课件处置：用修正版，并把"三个非递归遍历里只有后序版本漏了判空"当作一个易错点讲：写任何遍历函数，第一行先判空。

### A5：2024 判 BST 的 prev 初值与函数声明顺序（C2-B-05、C3-B-03）

问题：材料用 `int prev = -1` 作初值，隐含"所有结点值都大于 -1"的假设，树里有负数就错；另外 `JudgeBST` 调用 `checkBST` 时后者还没声明。

```c
// 修正版：prev 初值用 INT_MIN（或按题目给的下界），checkBST 先声明
#include <limits.h>
int checkBST(BiTNode *node, int &prev);      // 修正：先声明

int JudgeBST(BiTree T){
    int prev = INT_MIN;                      // 修正：不隐含"值都大于 -1"
    return checkBST(T, prev);
}
int checkBST(BiTNode *node, int &prev){
    if(node == NULL) return 1;
    if(!checkBST(node->lchild, prev)) return 0;      // 左子树不是 BST，短路
    if(node->data <= prev) return 0;                 // 中序序列必须严格递增
    prev = node->data;
    return checkBST(node->rchild, prev);
}
```

课件处置：用修正版，并把这个假设当教学点讲——**任何用"哨兵初值"的算法都要问一句：这个初值一定比所有合法数据小（或大）吗？** 如果题目保证了结点值都是正整数，写 `prev = 0` 也可以，但要在答案里写一句"设结点值均为正整数"。答题时把假设写出来是加分项，藏着不写是扣分点。

### A6：变长数组带初始化列表（C3-B-04）

问题：2025 真题第 3 题的解答写 `bool visited[G.vexnum] = {false};`，这是变长数组带初始化列表，与该文档自己的全局约定 5（只有 `={0}` 可作为伪代码简写）矛盾；同文档的 2024 第 2 题用循环赋 0，两种做法并存。

```c
// 修正版：固定长度数组加 ={false}，或变长数组加循环赋值，二选一并全课统一
bool visited[MaxVertexNum] = {false};        // 写法一：固定长度，可以用初始化列表

// 写法二：变长数组必须循环赋值
bool visited[G.vexnum];
for(int i = 0; i < G.vexnum; i++) visited[i] = false;
```

课件处置：统一用写法一（固定长度 MaxVertexNum），因为它与 A9 的常量统一口径一致，也不会踩约定 5 的坑。讲 2025 真题时在页面上说明一句："材料这里写的是变长数组带初始化，按我们的约定 5 应该改成固定长度或循环赋值"，正好把 A1 讲过的约定 5 再兑现一次。
